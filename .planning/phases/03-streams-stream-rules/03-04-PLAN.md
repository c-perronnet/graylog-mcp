    ```javascript
    // STREAM-08 — create_stream_rule. POST /api/streams/{streamId}/rules.
    //
    // D-09 parent-mutable pre-flight: GET /api/streams/{streamId}; refuse with
    // reason:stream_immutable BEFORE the POST when current.is_editable === false.
    // Built-in streams (system events streams) reject rule changes at the
    // server level too — this wrapper-side check saves the round-trip and
    // surfaces a structured reason.
    //
    // Pitfall S9: agent-facing API uses string discriminators; wire format is
    // numeric int via STREAM_RULE_TYPE_TO_NUMERIC.
    //
    // Pitfall S10: value/field/inverted are non-nullable on the wire. For
    // variants that semantically lack a field (always_match, match_input) or
    // a value (always_match, present), emit "" / false respectively.
    //
    // Pitfall S7: response is 201 + { streamrule_id }; normalize via
    // toIdBody with idFields: ["streamrule_id", "id"].

    import { defineMutatingHandler } from "../_shared/handler.js";
    import { CreateStreamRuleSchema, STREAM_RULE_TYPE_TO_NUMERIC } from "./schemas.js";
    import { makeClient } from "../../graylog/client.js";
    import { toIdBody } from "../../graylog/normalize.js";
    import { GraylogValidationError } from "../../graylog/errors.js";

    export const handleCreateStreamRule = defineMutatingHandler({
        name: "create_stream_rule",
        schema: CreateStreamRuleSchema,
        async build(args) {
            const client = makeClient(args._conn);
            const parentPath = `/api/streams/${args.streamId}`;

            // D-09 parent-mutable pre-flight.
            const parent = await client.request("GET", parentPath, null);
            if (parent.is_editable === false) {
                const err = new GraylogValidationError(
                    `Parent stream "${parent.title}" (id: ${args.streamId}) is non-editable; refusing rule creation. ` +
                    `Call list_streams and filter by mutable: true to find an editable candidate.`,
                    { status: 400, method: "POST", path: `${parentPath}/rules` },
                );
                err.reason = "stream_immutable";
                throw err;
            }

            // Pitfall S9 + S10: numeric translation + non-nullable wire defaults.
            const wireType = STREAM_RULE_TYPE_TO_NUMERIC[args.type];
            const wireBody = {
                type: wireType,
                value: String(args.value ?? ""),
                field: args.field ?? "",
                inverted: args.inverted ?? false,
                description: args.description ?? null,
            };

            return {
                method: "POST",
                path: `${parentPath}/rules`,
                body: wireBody,
                postApplyEstimate: { id: "__SERVER_ASSIGNED__" },
                normalize: (raw) => toIdBody(raw, { idFields: ["streamrule_id", "id"] }),
            };
        },
        apply: (client, req) => client.request(req.method, req.path, req.body),
        summarize: (args) => `Create ${args.type} rule on stream ${args.streamId}`,
    });
    ```

    ```javascript
    // rules have no cascade dependents (the rule itself is the leaf node;
    // deleting it does NOT cascade to other resources). NO confirmation hash,
    // NO requireConfirm gate.

    // check). NO cascade enumeration. build() returns a descriptor WITHOUT
    // a `cascades` key — handler.js's preview emitter spreads cascades only
    // when truthy (Plan 01-02 BLOCKER #1 amendment), so the absence of
    // `cascades` in the dry-run JSON is the structural assertion of leaf-
    // delete.
    //
    // Wire path: DELETE /api/streams/{streamId}/rules/{ruleId} → 204 No Content.
    // Sync (no system-job).

    import { defineMutatingHandler } from "../_shared/handler.js";
    import { DeleteStreamRuleSchema } from "./schemas.js";
    import { makeClient } from "../../graylog/client.js";
    import { GraylogValidationError } from "../../graylog/errors.js";

    export const handleDeleteStreamRule = defineMutatingHandler({
        name: "delete_stream_rule",
        schema: DeleteStreamRuleSchema,
        async build(args) {
            const client = makeClient(args._conn);
            const parentPath = `/api/streams/${args.streamId}`;

            // D-09 parent-mutable pre-flight (Pitfall S12).
            const parent = await client.request("GET", parentPath, null);
            if (parent.is_editable === false) {
                const err = new GraylogValidationError(
                    `Parent stream "${parent.title}" (id: ${args.streamId}) is non-editable; refusing rule deletion. ` +
                    `Call list_streams and filter by mutable: true.`,
                    { status: 400, method: "DELETE", path: `${parentPath}/rules/${args.ruleId}` },
                );
                err.reason = "stream_immutable";
                throw err;
            }

            // Leaf delete — NO cascades key set, NO _confirmationToken.
            return {
                method: "DELETE",
                path: `${parentPath}/rules/${args.ruleId}`,
                body: undefined,
                postApplyEstimate: { id: args.ruleId, deleted: true },
            };
        },
        apply: (client, req) => client.request(req.method, req.path, req.body),
        summarize: (args) => `Delete rule ${args.ruleId} from stream ${args.streamId}`,
    });
    ```

    ```javascript
    export const handleUpdateStreamRule = defineMutatingHandler({

        schema: UpdateStreamRuleSchema,
        async build(args) {
    ```
