```js
import { test } from "node:test";
import assert from "node:assert/strict";
import { readdirSync, readFileSync, statSync } from "node:fs";
import { join, dirname } from "node:path";
import { fileURLToPath } from "node:url";
import "./snapshot-config.js";

const __dirname = dirname(fileURLToPath(import.meta.url));
const SNAPSHOTS_DIR = join(__dirname, "__snapshots__");

// Patterns that, if present in a snapshot fixture, indicate a credential leak.
const DENY_PATTERNS = [
    { name: "Authorization header", regex: /Authorization/i },
    { name: "32+ char alphanumeric (apiToken-like)", regex: /[A-Za-z0-9]{32,}/ },
    { name: "password literal with value", regex: /password['"]?\s*[:=]\s*['"][^'"]+['"]/i },
];

// Allowlist of strings that LOOK like the pattern but are intentional fixtures.
// Examples: the literal "__SERVER_ASSIGNED__" sentinel, the test apiToken "_test",
// hex digests of deterministic sha-256 idempotency keys (these are 32 chars hex
// but they are derived from synthetic args, not real credentials).
//
// IMPORTANT — idempotency keys are 32 hex chars and WILL match the apiToken regex.
// The redaction-test must distinguish: a token in a SECRET context (after "apiToken:"
// or inside an "Authorization:" header) vs. an idempotency-key value. We handle
// this by ALLOWING the 32+ char pattern when the surrounding context is a known
// idempotencyKey field; otherwise we fail.

function isAllowedMatch(content, match, regex) {
    if (regex.source === /[A-Za-z0-9]{32,}/.source) {
        // Find the match location and inspect ~30 chars before it.
        const idx = content.indexOf(match);
        const context = content.slice(Math.max(0, idx - 40), idx);
        // Allowed contexts: idempotencyKey, _test apiToken (literally the string "_test" which is 5 chars, won't match anyway)
        if (/idempotencyKey['"]?\s*[:=]\s*['"]?$/.test(context)) return true;
    }
    return false;
}

test("no snapshot fixture contains Authorization header, apiToken-like strings, or password literals", () => {
    let entries;
    try {
        entries = readdirSync(SNAPSHOTS_DIR);
    } catch (err) {
        // Directory doesn't exist yet (no snapshots written). That's fine — there's nothing to check.
        // This test re-runs on every npm test invocation; once snapshots land, scan begins.
        return;
    }

    const snapshotFiles = entries.filter((name) => name.endsWith(".snapshot"));
    const violations = [];

    for (const file of snapshotFiles) {
        const fullPath = join(SNAPSHOTS_DIR, file);
        if (!statSync(fullPath).isFile()) continue;
        const content = readFileSync(fullPath, "utf8");
        for (const { name, regex } of DENY_PATTERNS) {
            const globalRegex = new RegExp(regex.source, regex.flags.includes("g") ? regex.flags : regex.flags + "g");
            let m;
            while ((m = globalRegex.exec(content)) !== null) {
                if (!isAllowedMatch(content, m[0], regex)) {
                    violations.push(`${file}: matches "${name}" → "${m[0].slice(0, 60)}..."`);
                }
            }
        }
    }

    assert.equal(
        violations.length,
        0,
        `Auth-redaction violations:\n${violations.join("\n")}`
    );
});
```

**Note on idempotency-key false positives:** The 32-char idempotency-key hex strings WILL match `/[A-Za-z0-9]{32,}/`. The `isAllowedMatch` function inspects the context preceding the match — if the surrounding text is `idempotencyKey":"` (with optional escaping), the match is whitelisted. This avoids the false-positive while still catching unscoped 32+ char tokens elsewhere.
