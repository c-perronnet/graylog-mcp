Graylog 7.0.6 extractor endpoint contracts (RESEARCH.md Endpoint Catalogue rows 9-12):
- GET /api/system/inputs/{inputId}/extractors → { extractors: ExtractorSummary[] } where ExtractorSummary = { id, title, cursor_strategy, source_field, target_field, extractor_type, extractor_config, converters, condition_type, condition_value, order, exceptions, metrics? }
- POST /api/system/inputs/{inputId}/extractors → CreateExtractorRequest body → 201 + Location + { extractor_id: "..." }
- PUT /api/system/inputs/{inputId}/extractors/{extractorId} → same body shape → 200 + full ExtractorSummary
- DELETE /api/system/inputs/{inputId}/extractors/{extractorId} → 204 No Content

Extractor common envelope (RESEARCH.md §"Extractor JSON Shapes" Example 5 lines 532-563):
```json
{
    "title": "Extract username from log message",
    "cursor_strategy": "copy",         // or "cut"
    "source_field": "message",
    "target_field": "username",
    "extractor_type": "grok",          // lowercase; one of 8: substring, regex, regex_replace, split_and_index, copy_input, grok, json, lookup_table
    "extractor_config": { ... },       // type-specific (see below)
    "converters": [ ... ],             // optional list of { type, config } pairs
    "condition_type": "none",          // or "string" or "regex"
    "condition_value": "",
    "order": 0
}
```

Per-type extractor_config shapes (RESEARCH.md lines 548-558):
| extractor_type    | extractor_config shape |
|-------------------|-------------------------|
| grok              | { grok_pattern: string, named_captures_only?: bool } |
| regex             | { regex_value: string } |
| regex_replace     | { regex: string, replacement: string, replace_all?: bool } |
| split_and_index   | { split_by: string, index: int } |
| substring         | { begin_index: int, end_index: int } |
| copy_input        | {} (empty) |
| json              | { list_separator?: string, key_separator?: string, kv_separator?: string, key_prefix?: string, key_whitespace_replacement?: string, replace_key_whitespace?: bool, flatten?: bool } |
| lookup_table      | { lookup_table_name: string } |

    ```js
    import { defineMutatingHandler } from "../_shared/handler.js";
    import { CreateExtractorSchema } from "./schemas.js";
    import { findExistingMatches } from "../_shared/conflict.js";
    import { resolveConnection } from "../_shared/connection.js";
    import { makeClient } from "../../graylog/client.js";
    import { toIdBody } from "../../graylog/normalize.js";

    export const handleCreateExtractor = defineMutatingHandler({
        name: "create_extractor",
        schema: CreateExtractorSchema,
        async build(args) {
            const seamArgs = args._testConnection ? { ...args, _testConnection: args._testConnection } : args;
            const { conn } = resolveConnection(seamArgs);
            const client = makeClient(conn);
            const existing = await findExistingMatches(client, {
                listPath: `/api/system/inputs/${args.inputId}/extractors`,
                matchFn: (item) => item.title === args.title && item.extractor_type === args.extractor_type,
                similarityReason: "exact title + extractor_type match",
            });
            return {
                method: "POST",
                path: `/api/system/inputs/${args.inputId}/extractors`,
                body: {
                    title: args.title,
                    cursor_strategy: args.cursor_strategy,
                    source_field: args.source_field,
                    target_field: args.target_field,
                    extractor_type: args.extractor_type,
                    extractor_config: args.extractor_config,
                    ...(args.converters !== undefined ? { converters: args.converters } : {}),
                    condition_type: args.condition_type,
                    condition_value: args.condition_value,
                    order: args.order,
                },
                postApplyEstimate: { id: "__SERVER_ASSIGNED__" },
                existingMatches: existing,
                normalize: (raw) => toIdBody(raw, { idFields: ["extractor_id", "id"] }),
            };
        },
        apply: (client, req) => client.request(req.method, req.path, req.body),
        summarize: (args) => `Create ${args.extractor_type} extractor "${args.title}" on input ${args.inputId}`,
    });
    ```

    ```js
    import { defineMutatingHandler } from "../_shared/handler.js";
    import { UpdateExtractorSchema } from "./schemas.js";
    import { resolveConnection } from "../_shared/connection.js";
    import { makeClient } from "../../graylog/client.js";
    import { toIdBody } from "../../graylog/normalize.js";

    export const handleUpdateExtractor = defineMutatingHandler({
        name: "update_extractor",
        schema: UpdateExtractorSchema,
        async build(args) {
            const seamArgs = args._testConnection ? { ...args, _testConnection: args._testConnection } : args;
            const { conn } = resolveConnection(seamArgs);
            const client = makeClient(conn);
            // Pre-flight: fetch current extractor for the merge
            const current = await client.request(
                "GET",
                `/api/system/inputs/${args.inputId}/extractors/${args.extractorId}`,
                null,
            );
            // Merge: changes win, everything else comes from current
            const mergedBody = {
                title: args.changes.title ?? current.title,
                cursor_strategy: args.changes.cursor_strategy ?? current.cursor_strategy,
                source_field: args.changes.source_field ?? current.source_field,
                target_field: args.changes.target_field ?? current.target_field,
                extractor_type: current.extractor_type,  // type is immutable
                extractor_config: args.changes.extractor_config ?? current.extractor_config,
                ...(args.changes.converters ?? current.converters
                    ? { converters: args.changes.converters ?? current.converters }
                    : {}),
                condition_type: args.changes.condition_type ?? current.condition_type,
                condition_value: args.changes.condition_value ?? current.condition_value,
                order: args.changes.order ?? current.order,
            };
            return {
                method: "PUT",
    ```
