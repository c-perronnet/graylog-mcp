    ```javascript
    // EVENT-05 — delete_event_definition. DELETE /api/events/definitions/{id}.
    //
    // D-08 informational cascade: pre-flights GET /api/events/definitions/{id}
    //   to read current.notifications[] and surface them in cascades.notifications
    //   in the dry-run preview. The agent sees WHICH notifications would lose
    //   the wiring before applying.
    //
    // CRITICAL — D-08 is INFORMATIONAL, NOT a refusal gate:
    //   - Notifications SURVIVE the event-def delete (they are independent
    //     resources — separate notification CRUD lives at Plan 05-04).
    //   - Only the WIRING (def → notification linkage) vanishes.
    //   - Therefore NO cascade-hash is issued; NO confirmationToken in the
    //     preview JSON; NO drift refusal at apply.
    //   - The dry-run cascades block is purely informational — mirrors the
    //     Phase 1 delete_input pattern (extractors enumerated but no token).
    //
    // Pitfall Q1 (Open Question 1 of RESEARCH): server-side EventResolver
    //   throws 400 ValidationException if other event_defs reference this one
    //   as a correlation parent. The wrapper does NOT pre-flight this because
    //   Phase 5's aggregation-v1-only scope doesn't construct correlation refs.
    //   Should it ever happen, the 400 surfaces via wrapGraylogError cleanly.
    //
    // Best-effort pre-flight: if GET fails (404 / 403), proceed with empty
    //   cascades array — the DELETE will surface the real error via the
    //   wrapper's apply() try/catch. Mirror Phase 1 delete_input lines 30-47.

    import { defineMutatingHandler } from "../_shared/handler.js";
    import { DeleteEventDefinitionSchema } from "./schemas.js";
    import { makeClient } from "../../graylog/client.js";

    export const handleDeleteEventDefinition = defineMutatingHandler({
        name: "delete_event_definition",
        schema: DeleteEventDefinitionSchema,
        async build(args) {
            const client = makeClient(args._conn);

            // D-08 best-effort informational pre-flight: enumerate notifications
            // the event def references. The DELETE call surfaces the real error
            // if pre-flight 404s (input doesn't exist) or 403s (permission denied).
            let notifications = [];
            try {
                const dto = await client.request(
                    "GET",
                    `/api/events/definitions/${args.definitionId}`,
                    null,
                );
                notifications = Array.isArray(dto?.notifications)
                    ? dto.notifications.map((n) => ({
                          notification_id: n.notification_id,
                          notification_parameters: n.notification_parameters ?? null,
                      }))
                    : [];
            } catch (_err) {
                notifications = [];
            }

            return {
                method: "DELETE",
                path: `/api/events/definitions/${args.definitionId}`,
                body: undefined,
                postApplyEstimate: { id: args.definitionId, deleted: true },
                // D-08: INFORMATIONAL cascade — no confirmationToken issued.
                cascades: { notifications },
            };
        },
        apply: (client, req) => client.request(req.method, req.path, req.body),
        summarize: (args) =>
            `Delete event definition ${args.definitionId} (any notifications referenced lose this def's wiring; notifications themselves survive)`,
    });
    ```
