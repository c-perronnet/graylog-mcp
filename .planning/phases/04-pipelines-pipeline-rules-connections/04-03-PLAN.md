<!-- D-11 Structured intent schema (RESEARCH §"Structured Intent Grammar" lines 773-825) -->
RuleSpec = { name: string, when: Condition, then: Action[] }

Condition = | { type:"comparison", op, left:Expression, right:Expression }
           | { type:"and",        left:Condition, right:Condition }
           | { type:"or",         left:Condition, right:Condition }
           | { type:"not",        expr:Condition }
           | { type:"function_call", name, args:{positional?|named?} }
           | { type:"has_field",  field:string }
           | { type:"field_ref",  field, source?:"message"|{type,name} }
           | { type:"literal",    value:string|number|boolean|null }
Expression = same union as Condition (RuleLang.g4 treats them as one non-terminal)

Action = | { type:"set_field", field, value, ...optional }
        | { type:"remove_field", field, message? }
        | { type:"rename_field", old_field, new_field, message? }
        | { type:"lookup_value", target_field, lookup_table, key, default? }
        | { type:"function_call_statement", name, args }
        | { type:"let_assignment", var_name, value }

    // D-11 Structured Intent Grammar (RESEARCH §"Structured Intent Grammar" lines

        type: z.literal("literal"),
        value: z.union([z.string(), z.number(), z.boolean(), z.null()]),
    });

    const FieldRefSchema = z.object({
        type: z.literal("field_ref"),
        field: z.string().min(1),
        source: z.union([
            z.literal("message"),
            z.object({ type: z.literal("identifier"), name: z.string().min(1) }),

        type: z.literal("has_field"),
        field: z.string().min(1),
    });

    export const FunctionCallSchema = z.lazy(() =>

            type: z.literal("function_call"),
            name: z.string().min(1),
            args: z.object({
                positional: z.array(ExpressionSchema).optional(),
                named: z.record(ExpressionSchema).optional(),

    // ExpressionSchema = Condition variants treated as expressions
    // (RuleLang.g4 line 77-96 — single non-terminal).
    export const ExpressionSchema = z.lazy(() => z.union([
        LiteralSchema,
        FieldRefSchema,
        HasFieldSchema,
        FunctionCallSchema,
        ConditionSchema,
    ]));

    export const ConditionSchema = z.lazy(() => z.discriminatedUnion("type", [
        z.object({
            type: z.literal("comparison"),
            op: z.enum(["==", "!=", "<", "<=", ">", ">="]),
            left: ExpressionSchema,
            right: ExpressionSchema,
        }),
        z.object({ type: z.literal("and"), left: ConditionSchema, right: ConditionSchema }),
        z.object({ type: z.literal("or"),  left: ConditionSchema, right: ConditionSchema }),
        z.object({ type: z.literal("not"), expr: ConditionSchema }),
        FunctionCallSchema,
        HasFieldSchema,
        FieldRefSchema,
        LiteralSchema,
    ]));

        z.object({
            type: z.literal("set_field"),
            field: z.string().min(1),
            value: ExpressionSchema,
            message: ExpressionSchema.optional(),
            default: ExpressionSchema.optional(),
            clean_field: z.boolean().optional(),
            prefix: z.string().optional(),
            suffix: z.string().optional(),
        }),
        z.object({
            type: z.literal("remove_field"),
            field: z.string().min(1),

            type: z.literal("rename_field"),
            old_field: z.string().min(1),
            new_field: z.string().min(1),
            message: ExpressionSchema.optional(),

            target_field: z.string().min(1),
            lookup_table: z.string().min(1),

            name: z.string().min(1),
            args: z.object({

                named: z.record(ExpressionSchema).optional(),

        }),

            type: z.literal("let_assignment"),
            var_name: z.string().min(1),
            value: ExpressionSchema,
        }),
    ]);

    export const RuleSpecSchema = z.object({
        name: z.string().min(1),
        when: ConditionSchema,
        then: z.array(ActionSchema).min(1),  // at least one action
    });
