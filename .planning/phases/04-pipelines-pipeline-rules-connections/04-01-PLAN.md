    **C. Create `src/pipeline-dsl/escape.js`** — per RESEARCH §"DSL String Escape Character Set (D-13)" lines 960-985 verbatim recommended implementation:
    ```js
    // Pipeline-rule DSL escape helpers (D-12 / D-13). Pure CPU; no I/O.
    // Defense-in-depth backstop is the server-side parse pre-flight (D-05) —
    // every embedded literal in an emitted rule routes through escapeString
    // first, AND the wrapper sends the emitted source through
    // POST /system/pipelines/rule/parse before any apply.
    //
    // Character set verified against source-code/.../RuleLang.g4:363-381.

    const SHORT = {
        0x08: "\\b",  0x09: "\\t",  0x0A: "\\n",  0x0C: "\\f",
        0x0D: "\\r",  0x22: '\\"',  0x27: "\\'",  0x5C: "\\\\",
    };

    export function escapeString(s) {
        if (typeof s !== "string") {
            throw new TypeError(`escapeString: expected string, got ${typeof s}`);
        }
        let out = "";
        for (const ch of s) {
            const code = ch.codePointAt(0);
            if (SHORT[code] !== undefined) { out += SHORT[code]; continue; }
            if (code < 0x20) {
                out += "\\u" + code.toString(16).padStart(4, "0");
                continue;
            }
            out += ch;
        }
        return out;
    }

    export function escapeValue(v) {
        // Type-aware DSL literal emission. Objects, arrays, undefined are
        // refused — the agent must compose them through structured intent
        // FunctionCall / FieldRef nodes, NOT inline as a JS value.
        if (v === null) return "null";
        if (typeof v === "string") return `"${escapeString(v)}"`;
        if (typeof v === "number" || typeof v === "boolean") return String(v);
        throw new TypeError(
            `escapeValue: refusing to emit ${typeof v} as a DSL literal — use structured intent FunctionCall / FieldRef instead`,
        );
    }
    ```

    ```js
    function liveSignatureString(live) {

        // Render TS-ish signature: fn(arg1: type, arg2?: type): returnType
    ```

    ```js
    // Client-side validator for pipeline-rule DSL source (D-04).
    //
    // CONSUMES the MERGED Map from function-catalogue.js — NOT the raw
    // staticBuiltins array. Pitfall 5 (RESEARCH §"Pitfall 5: Hand-curated
    // builtins.js drifts"): a live-only function name MUST validate. If
    // we read from staticBuiltins, Graylog 7.2.x's new functions would
    // false-fail before the parse pre-flight runs.
    //
    // Strictness: function-name + paren-balance ONLY. Arg-count is OUT OF
    // SCOPE for this phase (warning-only per RESEARCH Open Question #2 —
    // server parse pre-flight is the authoritative gate; lint is informational).

    export function validateRuleSource(source, mergedCatalogue) {
        const errors = [];
        const warnings = [];

        if (typeof source !== "string" || source.length === 0) {
            warnings.push({ type: "empty_source", message: "rule source is empty" });
            return { errors, warnings };
        }

        // Paren balance — naive single-pass counter (ignores strings; the
        // server parser is authoritative for string-aware balance).
        let parens = 0;
        for (const ch of source) {
            if (ch === "(") parens += 1;
            else if (ch === ")") parens -= 1;
            if (parens < 0) {
                errors.push({ type: "paren_imbalance", message: `unbalanced ) at offset ${source.indexOf(ch)}` });
                break;
            }
        }
        if (parens > 0) {
            errors.push({ type: "paren_imbalance", message: `${parens} unmatched (` });
        }

        // Function-name lint. Regex: word followed by ( — captures the name.
        // Skip Graylog reserved keywords (rule, when, then, end, let, true,
        // false, null, has_field is_null is_not_null) that look like function
        // calls but are grammar productions (has_field IS in the catalogue;
        // the reserved-list filter handles bare identifiers, not function
        // application).
        const RESERVED = new Set(["if", "true", "false", "null"]);
        const fnPattern = /\b([a-z_][a-z0-9_]*)\s*\(/gi;
        let m;
        while ((m = fnPattern.exec(source)) !== null) {
            const name = m[1];
            if (RESERVED.has(name)) continue;
            if (!mergedCatalogue.has(name)) {
                errors.push({
                    type: "unknown_function",
                    message: `unknown function "${name}" (not in merged catalogue) at offset ${m.index}`,
                    name,
                    offset: m.index,
                });
            }
        }

        return { errors, warnings };
    }
    ```
