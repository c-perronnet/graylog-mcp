```javascript
import { defineMutatingHandler } from "../_shared/handler.js";
import { SetupLongTermArchivalIndexSchema } from "./schemas.js";
import { createIndexSet } from "../../services/index-sets.js";
import { executeChain } from "../_shared/blueprint-chain.js";
import { SERVER_ASSIGNED_SENTINEL } from "../_shared/dry-run.js";

const ONE_GIB = 1_073_741_824;

function slugify(s) {
  return s.toLowerCase().replace(/[^a-z0-9]+/g, "_").replace(/^_|_$/g, "");
}

export const handleSetupLongTermArchivalIndex = defineMutatingHandler({
  name: "setup_long_term_archival_index",
  schema: SetupLongTermArchivalIndexSchema,
  async build(args) {
    const indexPrefix = args.indexPrefix ?? slugify(args.name);
    const indexSetBody = {
      title: args.name,
      description: args.description ?? `Long-term archival index for ${args.name}`,
      index_prefix: indexPrefix,
      shards: args.shards ?? 4,
      replicas: args.replicas ?? 1,
      rotation_strategy_class: "org.graylog2.indexer.rotation.strategies.SizeBasedRotationStrategyConfig",
      rotation_strategy: {
        type: "org.graylog2.indexer.rotation.strategies.SizeBasedRotationStrategyConfig",
        max_size: ONE_GIB,
      },
      retention_strategy_class: "org.graylog2.indexer.retention.strategies.DeletionRetentionStrategyConfig",
      retention_strategy: {
        type: "org.graylog2.indexer.retention.strategies.DeletionRetentionStrategyConfig",
        max_number_of_indices: args.retentionDays,  // approximation: 1 index per day
      },
      writable: true,
      index_analyzer: "standard",
      index_optimization_max_num_segments: 1,
      index_optimization_disabled: false,
      field_type_refresh_interval: 5000,
    };

    const chain = [{
      step: 1,
      tool: "create_index_set",
      request: {
        method: "POST",
        path: "/api/system/indices/index_sets",
        body: indexSetBody,
      },
      postApplyEstimate: {id: SERVER_ASSIGNED_SENTINEL},
    }];

    return {
      chain,
      method: "POST",
      path: "/api/system/indices/index_sets",
      body: indexSetBody,
      postApplyEstimate: {id: SERVER_ASSIGNED_SENTINEL},
    };
  },
  async apply(client, req) {
    const result = await executeChain(client, req.chain);
    if (result.isError) return result;
    return result.transcript[0].response;
  },
  summarize: (args) => `Set up long-term archival index "${args.name}" with ${args.retentionDays}-day retention`,
});
```

```javascript
import { defineMutatingHandler } from "../_shared/handler.js";
import { SetupDebugLogDroppingSchema } from "./schemas.js";
import { createPipeline, createRule, connectToStream } from "../../services/pipelines.js";
import { executeChain } from "../_shared/blueprint-chain.js";
import { emitRule } from "../../pipeline-dsl/emit.js";
import { SERVER_ASSIGNED_SENTINEL } from "../_shared/dry-run.js";

export const handleSetupDebugLogDropping = defineMutatingHandler({
  name: "setup_debug_log_dropping",
  schema: SetupDebugLogDroppingSchema,
  async build(args) {
    const ruleTitle = args.ruleTitle ?? `drop_sub_${args.minLevel}`;
    const pipelineTitle = args.pipelineTitle ?? `Drop sub-${args.minLevel} for stream ${args.streamId}`;

    // Step 1: createRule via emitRule
    const ruleSource = emitRule({
      name: ruleTitle,
      when: {
        type: "comparison",
        left: {type: "field_ref", source: "message", field: "level"},
        op: ">",
        right: {type: "literal", value: args.minLevel},
      },
      then: [{type: "function_call_statement", name: "drop_message", args: {positional: []}}],
    });

    // Step 2: createPipeline — stage references the rule by title (Phase 4 contract).
    // Reference verification: see src/tools/pipelines/create-pipeline.js precedent.
    const pipelineSource = [
      `pipeline "${pipelineTitle}"`,
      `stage 0 match either`,
      `    rule "${ruleTitle}"`,
      `end`,
    ].join("\n");

    const chain = [
      {
        step: 1,
        tool: "create_pipeline_rule",
        request: {
          method: "POST",
          path: "/api/system/pipelines/rule",
          body: {title: ruleTitle, description: `Drop messages where level > ${args.minLevel}`, source: ruleSource},
        },
        postApplyEstimate: {id: SERVER_ASSIGNED_SENTINEL},
      },
      {
        step: 2,
        tool: "create_pipeline",
        request: {
          method: "POST",
          path: "/api/system/pipelines/pipeline",
          body: {title: pipelineTitle, description: `Pipeline that drops sub-${args.minLevel} for stream ${args.streamId}`, source: pipelineSource},
        },
        // No dependsOn — the pipeline references the rule by TITLE in its source string,
        // not by ID. The createRule must succeed FIRST sequentially, but no placeholder substitution is needed.
        postApplyEstimate: {id: SERVER_ASSIGNED_SENTINEL},
      },
      {
        step: 3,
        tool: "connect_pipelines_to_stream",
        request: {
          method: "POST",
          path: "/api/system/pipelines/connections/to_stream",
          body: {stream_id: args.streamId, pipeline_ids: [`${SERVER_ASSIGNED_SENTINEL}step2`]},
        },
        dependsOn: {from: "step2.response.id", as: "pipeline_ids[0]"},
      },
    ];

    return {
      chain,
      method: "POST",
      path: "/api/system/pipelines/connections/to_stream",  // primary preview = final step
      body: chain[2].request.body,
      postApplyEstimate: {id: args.streamId},
    };
  },
  async apply(client, req) {
    const result = await executeChain(client, req.chain);
    if (result.isError) return result;
    return result.transcript[result.transcript.length - 1].response;
  },
  summarize: (args) => `Drop messages with level > ${args.minLevel} on stream ${args.streamId} (3-step chain: rule + pipeline + connect)`,
});
```

**Step B — `src/tools/blueprints/setup-pipeline-for-stream.js`**:

```javascript
import { defineMutatingHandler } from "../_shared/handler.js";
import { SetupPipelineForStreamSchema } from "./schemas.js";
import { createPipeline, createRule, connectToStream } from "../../services/pipelines.js";
import { executeChain } from "../_shared/blueprint-chain.js";
import { emitRule } from "../../pipeline-dsl/emit.js";
import { SERVER_ASSIGNED_SENTINEL } from "../_shared/dry-run.js";

export const handleSetupPipelineForStream = defineMutatingHandler({
  name: "setup_pipeline_for_stream",
  schema: SetupPipelineForStreamSchema,
  async build(args) {
    // Step 1..N: create one rule per transform via emitRule
    const ruleSteps = args.transforms.map((spec, i) => {
      const source = emitRule(spec);
      return {
        step: i + 1,
        tool: "create_pipeline_rule",
        request: {
          method: "POST",
          path: "/api/system/pipelines/rule",
          body: {title: spec.name, description: `Generated by setup_pipeline_for_stream`, source},
        },
        postApplyEstimate: {id: SERVER_ASSIGNED_SENTINEL},
      };
    });

    const N = args.transforms.length;

    // Step N+1: createPipeline. Source references each rule by title.
    const stageLines = args.transforms.map(t => `    rule "${t.name}"`);
    const pipelineSource = [
      `pipeline "${args.pipelineTitle}"`,
      `stage 0 match either`,
      ...stageLines,
      `end`,
    ].join("\n");
    const pipelineStep = {
      step: N + 1,
      tool: "create_pipeline",
      request: {
        method: "POST",
        path: "/api/system/pipelines/pipeline",
        body: {title: args.pipelineTitle, description: args.pipelineDescription ?? "", source: pipelineSource},
      },
      // Rules must exist FIRST sequentially; the source string carries titles so no apply-time placeholder substitution is needed.
      postApplyEstimate: {id: SERVER_ASSIGNED_SENTINEL},
    };

    // Step N+2: connectToStream — depends on step N+1's pipeline id.
    const connectStep = {
      step: N + 2,
      tool: "connect_pipelines_to_stream",
      request: {
        method: "POST",
        path: "/api/system/pipelines/connections/to_stream",
        body: {stream_id: args.streamId, pipeline_ids: [`${SERVER_ASSIGNED_SENTINEL}step${N + 1}`]},
      },
      dependsOn: {from: `step${N + 1}.response.id`, as: "pipeline_ids[0]"},
    };

    const chain = [...ruleSteps, pipelineStep, connectStep];

    return {
      chain,
      method: "POST",
```
